\section{第一章：网络效应}

\StudyTerm{\PriorityMust}{网络效应}{新增用户改变其他用户从产品或服务中获得的价值，属于消费外部性。}
正向效应来自互补，负向效应可来自拥挤；“组内/跨组”与“正向/负向”是不同的分类维度。

\StudyTerm{\PriorityImportant}{规模经济}{产量增加使平均资源成本下降，作用在供给侧。}
与网络效应的区别：规模经济降低成本，网络效应改变用户价值。

\StudyPoint{\PriorityMust}{社会边际收益}\label{kp:marginal-benefit}
\begin{proofbox}[title=推导]
设网络规模 $N$ 连续，人均毛收益 $U(N)$ 可微，单位资源成本 $c$ 恒定。
\begin{align*}
  W(N)&=NU(N)-cN,\\
  W'(N)&=\underbrace{U(N)}_{\text{新增者的毛收益}}
    +\underbrace{NU'(N)}_{\text{既有用户的外部收益}}-c.
\end{align*}
其中 $NU'(N)$ 是新增参与者带给既有用户的外部收益。此式是边际收益分解，不单凭 $W'(N)=0$ 断言全局最优。
\end{proofbox}

\section{第二章：网络规模计算}

\StudyPoint{\PriorityMust}{线性网络效应下的均衡规模}\label{kp:network-size}
\begin{examplebox}[title=例题]
\textbf{题目：}100 位用户的内在价值 $b$ 均匀分布于 $[0,100]$，加入效用为 $b-40+0.3n$。按连续规模求参与均衡。

\textbf{解：}边际用户满足 $\hat b-40+0.3n=0$，且 $n=100-\hat b$。
\begin{align*}
  n&=100-(40-0.3n)=60+0.3n,\\
  n^*&=\frac{60}{0.7}\approx85.71,\qquad
  \hat b^*=40-0.3n^*\approx14.29.
\end{align*}
阈值在 $[0,100]$ 内，内点解可行。$n=0$ 不是均衡：$b>40$ 的用户即使没有网络收益仍愿加入。
